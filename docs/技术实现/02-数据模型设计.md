## 一、核心数据模型

### 1.1 AppInfo — 应用信息模型

```swift
/// 单个应用的完整信息，扫描后不可变
struct AppInfo: Identifiable, Hashable, Codable {
    let id: String              // 等于 bundleID，唯一键
    let bundleID: String
    let displayName: String     // 优先 CFBundleDisplayName
    let url: URL                // .app 文件路径
    let version: String         // CFBundleShortVersionString
    let isMASApp: Bool          // 是否含 _MASReceipt
    let categoryType: String?   // LSApplicationCategoryType（用于文件夹自动命名）
    
    // 图标不存储在模型中，通过 IconCache 懒加载
}
```

**设计说明**：
- `id` 使用 `bundleID` 而非路径，路径可能因移动/更新变化，bundleID 稳定
- 图标（`NSImage`）不放入模型，避免 Codable 序列化问题和内存浪费

---

### 1.2 FolderInfo — 文件夹模型

```swift
/// 用户创建的应用分组文件夹
struct FolderInfo: Identifiable, Hashable, Codable {
    let id: UUID
    var name: String                // 用户自定义名称
    var appIDs: [String]            // 文件夹内应用的 bundleID 顺序列表
    var isUserNamed: Bool           // 是否被用户手动命名过（影响自动命名逻辑）
}
```

---

### 1.3 LayoutItem — 网格槽位元素

```swift
/// 网格中每个槽位放置的内容（应用 or 文件夹）
enum LayoutItem: Hashable, Codable {
    case app(bundleID: String)
    case folder(id: UUID)
}
```

---

### 1.4 LayoutData — 完整布局数据

```swift
/// 所有页面的图标排列布局
struct LayoutData: Codable {
    /// 二维数组：pages[pageIndex][slotIndex] = LayoutItem
    var pages: [[LayoutItem]]
    
    /// 文件夹信息字典
    var folders: [UUID: FolderInfo]
    
    /// 版本号，用于未来格式迁移
    var version: Int = 1
}
```

**布局规则**：
- `pages[0][0]` 是第一页第一个槽位（从左到右、从上到下）
- 槽位数 = 列数 × 5行，多余应用自动到下一页
- 文件夹视为占用 1 个槽位

### 1.5 UserPreferences — 用户偏好设置

```swift
/// 所有用户可配置的偏好项
struct UserPreferences: Codable {
    // 行为设置
    var focusSearchOnOpen: Bool = false
    var multiMonitorBehavior: MultiMonitorBehavior = .primaryScreen
    var showMenuBarIcon: Bool = false
    
    // 快捷键（存储为 Key+Modifier 的编码字符串）
    var triggerHotkey: HotkeyConfig = HotkeyConfig(keyCode: 0x7E, modifiers: [])  // F4
    
    // 外观设置
    var columnCountOverride: Int? = nil  // nil = 自动根据屏幕宽度计算
    var iconSizeOverride: CGFloat? = nil // nil = 默认 80pt
    var backgroundBlurRadius: CGFloat = 20
    
    enum MultiMonitorBehavior: String, Codable, CaseIterable {
        case primaryScreen = "primary"
        case mouseScreen = "mouse"
    }
    
    struct HotkeyConfig: Codable {
        var keyCode: UInt16
        var modifiers: NSEvent.ModifierFlags
        
        // ModifierFlags 不直接支持 Codable，需自定义编码
        enum CodingKeys: String, CodingKey { case keyCode, modifiersRaw }
        func encode(to encoder: Encoder) throws { /* ... */ }
        init(from decoder: Decoder) throws { /* ... */ }
    }
}
```

## 二、ViewModel 状态设计

### 2.1 LaunchpadViewModel

```swift
@Observable
@MainActor
final class LaunchpadViewModel {

    var searchText: String = ""

    var isEditMode: Bool = false

    var dragState: DragState = DragState()
    
    // ── 后台服务（依赖注入） ──────────────────────────────
```

### 2.2 DragState — 拖拽状态

```swift
struct DragState {
    var isDragging: Bool = false
    var draggedItemID: LayoutItem? = nil    // 被拖拽的图标/文件夹
    var dragLocation: CGPoint = .zero
    var sourcePageIndex: Int = 0
    var sourceSlotIndex: Int = 0
    
    // 实时计算的目标位置（用于其他图标让位动画）
    var targetPageIndex: Int = 0
    var targetSlotIndex: Int = -1
    
    // 悬停在文件夹上时高亮
    var hoveringFolderID: UUID? = nil
    
    // 边缘翻页
    var edgeScrollDirection: EdgeDirection? = nil
    
    enum EdgeDirection { case previous, next }
    
    var isEmpty: Bool { !isDragging }
}
```

## 三、计算属性设计

### 3.1 搜索结果

```swift
extension LaunchpadViewModel {
    /// 当前搜索词下的匹配应用
    var searchResults: [AppInfo] {
        guard !searchText.isEmpty else { return [] }
        let query = searchText.lowercased()
        return allApps
            .filter { app in
                app.displayName.lowercased().contains(query) ||
                app.bundleID.lowercased().contains(query)
            }
            .sorted { a, b in
                // 前缀匹配优先
                let aPrefix = a.displayName.lowercased().hasPrefix(query)
                let bPrefix = b.displayName.lowercased().hasPrefix(query)
                if aPrefix != bPrefix { return aPrefix }
                return a.displayName < b.displayName
            }
    }
    
    /// 当前是否处于搜索模式
    var isSearching: Bool { !searchText.isEmpty }
}
```

### 3.2 当前页图标

```swift
extension LaunchpadViewModel {
    /// 当前页的布局项（含拖拽中的实时重排）
    var currentPageItems: [LayoutItem] {
        guard currentPageIndex < layout.pages.count else { return [] }
        
        if dragState.isDragging {
            // 应用拖拽中的临时布局变化
            return applyDragReorder(to: layout.pages[currentPageIndex])
        }
        return layout.pages[currentPageIndex]
    }
    
    /// 总页数
    var totalPages: Int { layout.pages.count }
}
```

### 3.3 网格布局参数

```swift
extension LaunchpadViewModel {
    func columnCount(for screen: NSScreen) -> Int {
        if let override = preferences.columnCountOverride { return override }
        switch screen.frame.width {
        case 1440...: return 7
        case 1280..<1440: return 6
        default: return 5
        }
    }
    
    var iconSize: CGFloat {
        preferences.iconSizeOverride ?? 80
    }
    
    var itemsPerPage: Int {
        columnCount(for: NSScreen.main ?? NSScreen.screens[0]) * 5
    }
}
```

## 四、持久化策略

### 4.1 LayoutData 持久化

```
存储格式：JSON（通过 Codable 序列化）
存储路径：~/Library/Application Support/Launchpad/layout.json
写入时机：
  - 拖拽排序松开后（延迟 0.3s，合并连续操作）
  - 文件夹创建/重命名/删除后（立即写入）
  - 应用卸载后自动清理（立即写入）
读取时机：
  - App 启动时
  - FSEvents 触发重新扫描后（合并新应用，保留已有顺序）
```

### 4.2 布局合并逻辑（新应用加入）

```
已有布局 + 新扫描到的应用列表
    │
    ├─ 计算差集：新应用 = 扫描列表 - 布局中已有
    ├─ 计算差集：已卸载 = 布局中已有 - 扫描列表
    │
    ├─ 从布局中移除已卸载应用（其他图标前移补位）
    └─ 新应用追加到最后一页末尾（或新建页）
```

### 4.3 UserPreferences 持久化

```
存储格式：UserDefaults（Domain: com.yourcompany.AppLaunchpad）
写入时机：设置项变化后立即写入（property observer）
```

## 五、数据流时序图

### 5.1 首次启动数据流

```
App 启动
    │
    ├─ LayoutStore.load() → LayoutData（可能为空）
    ├─ UserPreferences.load() → preferences
    │
    └─ AppScanner.scan() [async, 后台]
            │
            ▼（扫描完成）
       LaunchpadViewModel.allApps = scannedApps
       LaunchpadViewModel.layout = mergeLayout(existing, scanned)
            │
            ▼
       UI 就绪（首次呼出可正常展示）
```

### 5.2 应用安装后的数据流

```
用户安装新 App
    │
    ▼
FSEvents 回调（/Applications 目录变化）
    │ debounce 2s
    ▼
AppScanner.scan() [async]
    │
    ▼
mergeLayout(existing: layout, scanned: newApps)
    │
    ▼（新 App 追加到末页）
LaunchpadViewModel.layout 更新
    │
    ▼
LayoutStore.save() + UI 刷新
```
